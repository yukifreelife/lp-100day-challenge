- [ ] **Step 3: Before/After JavaScriptを追加**

`js/script.js` に以下を追加:

```javascript
// ===================================
// Before/After Slider
// ===================================
const baSlider = document.querySelector('.ba-slider-container');
const baHandle = document.querySelector('.ba-handle');
const baBefore = document.querySelector('.ba-before');
const baAfter = document.querySelector('.ba-after');
const baNavBtns = document.querySelectorAll('.ba-nav-btn');
const baCounter = document.querySelector('.ba-counter');

if (baSlider && baHandle) {
  let isDown = false;

  baHandle.addEventListener('mousedown', () => isDown = true);
  document.addEventListener('mouseup', () => isDown = false);
  
  baSlider.addEventListener('mousemove', (e) => {
    if (!isDown) return;
    updateSlider(e.clientX);
  });
  
  baHandle.addEventListener('touchstart', () => isDown = true);
  document.addEventListener('touchend', () => isDown = false);
  
  baSlider.addEventListener('touchmove', (e) => {
    if (!isDown) return;
    updateSlider(e.touches[0].clientX);
  });
  
  function updateSlider(clientX) {
    const rect = baSlider.getBoundingClientRect();
    let x = clientX - rect.left;
    x = Math.max(0, Math.min(x, rect.width));
    const percent = (x / rect.width) * 100;
    
    baBefore.style.clipPath = `inset(0 ${100 - percent}% 0 0)`;
    baAfter.style.clipPath = `inset(0 0 0 ${percent}%)`;
    baHandle.style.left = `${percent}%`;
  }
}
```
